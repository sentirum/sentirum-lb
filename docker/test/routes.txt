# Static routes for local testing
# These are loaded FIRST and have highest priority

# Web app — two backends for round-robin
route add web-blue myhost.com/ http://upstream-web-blue:8080/
route add web-green myhost.com/ http://upstream-web-green:8081/

# API — single backend
route add api myhost.com/api/ http://upstream-api:9090/

# Static — with path strip
route add static static.example.com/ http://upstream-static:3000/ opts "strip=/static"

# Catch-all for no-host routes
route add catchall / http://upstream-web-blue:8080/
